cmake_minimum_required(VERSION 3.16)
project(europa_hostlink CXX)
set(CMAKE_CXX_STANDARD 17)
# PLEXIL Release 6 install (docs/baselines/plexil.md)
set(PLEXIL_HOME $ENV{HOME}/Programs/plexil CACHE PATH "PLEXIL installation")
find_library(PLEXIL_APP PlexilAppFramework PATHS ${PLEXIL_HOME}/lib REQUIRED NO_DEFAULT_PATH)
find_library(PLEXIL_INTFC PlexilIntfc PATHS ${PLEXIL_HOME}/lib REQUIRED NO_DEFAULT_PATH)
find_library(PLEXIL_VALUE PlexilValue PATHS ${PLEXIL_HOME}/lib REQUIRED NO_DEFAULT_PATH)
find_library(PLEXIL_UTILS PlexilUtils PATHS ${PLEXIL_HOME}/lib REQUIRED NO_DEFAULT_PATH)
# universalExec's DynamicLoader: libHostLink.so, initHostLink()
add_library(HostLink SHARED HostLinkAdapter.cc)
target_include_directories(HostLink PRIVATE ${PLEXIL_HOME}/include)
target_link_libraries(HostLink ${PLEXIL_APP} ${PLEXIL_INTFC} ${PLEXIL_VALUE} ${PLEXIL_UTILS})
